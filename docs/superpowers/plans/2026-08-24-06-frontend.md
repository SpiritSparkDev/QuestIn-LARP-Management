# Frontend Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A working Vanilla-JS frontend — login/register/email-verify/password-reset, account management, character creation with a per-event dynamic form, and admin pages for event/schema management and check-in — wired against the finished backend API. This is the last of the six plans; after this the app is usable end to end.

**Architecture:** Multiple standalone HTML pages under `frontend/`, each with an inline `<script type="module">` that calls a tiny shared `api.js` fetch wrapper. No build step, no bundler, no frontend framework — the browser loads ES modules natively. A new backend piece is required first: **none of the previous four plans built static file serving**, so `GET /login.html` etc. currently 404. Task 1 adds a small, path-traversal-safe static file handler as a fallback in `backend/server.js` when no API route matches.

**Tech Stack:** No new dependencies, front or back.

**Spec:** [docs/superpowers/specs/2026-08-24-teilnehmerregistrierung-design.md](../specs/2026-08-24-teilnehmerregistrierung-design.md)

**Builds on:** All five prior plans (merged) — the complete REST API: `/auth/*`, `/account`, `/events*`, `/characters*`, `/events/:id/register`, `/registrations`, `/events/:id/participants`, `/events/:id/checkin`, `/events/:id/checkout`.

## Global Constraints

- Kein Build-Step, kein Bundler — Browser laden ES-Module nativ.
- Mehrere eigenständige HTML-Seiten statt SPA-Router; jede lädt ein `<script type="module">`.
- Kein Frontend-Framework.
- Rollenprüfung bleibt serverseitig — das Frontend zeigt Navigation/Formulare, verlässt sich aber nicht darauf, dass ein Nutzer sie nicht sieht; ein 401/403 von der API wird angezeigt bzw. zum Login umgeleitet.
- Pure, DOM-freie Logik (API-Client, Formularfeld-Rendering) wird mit `node:test` unit-getestet; die HTML-Seiten selbst werden manuell im Browser verifiziert (kein Test-Framework für DOM-Tests im Stack).

- [ ] **Step 3: Write `backend/staticFiles.js`**

```js
import { readFile } from 'node:fs/promises';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

const DEFAULT_FRONTEND_DIR = path.join(path.dirname(fileURLToPath(import.meta.url)), '..', 'frontend');

const CONTENT_TYPES = {
  '.html': 'text/html; charset=utf-8',
  '.css': 'text/css; charset=utf-8',
  '.js': 'text/javascript; charset=utf-8',
};

export async function serveStaticFile(pathname, baseDir = DEFAULT_FRONTEND_DIR) {
  const relativePath = pathname === '/' ? '/index.html' : pathname;
  const resolvedBase = path.resolve(baseDir);
  const resolved = path.resolve(resolvedBase, `.${relativePath}`);
  if (resolved !== resolvedBase && !resolved.startsWith(resolvedBase + path.sep)) {
    return null;
  }

  const ext = path.extname(resolved);
  const contentType = CONTENT_TYPES[ext];
  if (!contentType) return null;

  try {
    const data = await readFile(resolved);
    return { data, contentType };
  } catch {
    return null;
  }
}
```

- [ ] **Step 3: Write `frontend/js/api.js`**

```js
async function request(path, options = {}) {
  const res = await fetch(path, {
    ...options,
    credentials: 'include',
    headers: { 'Content-Type': 'application/json', ...options.headers },
  });

  let body = null;
  try {
    body = await res.json();
  } catch {
    // no/invalid JSON body — fine for 204s and similar
  }

  if (!res.ok) {
    const error = new Error(body?.error || `request failed with status ${res.status}`);
    error.status = res.status;
    error.body = body;
    throw error;
  }

  return body;
}

export const api = {
  get: (path) => request(path),
  post: (path, data) => request(path, { method: 'POST', body: JSON.stringify(data) }),
  put: (path, data) => request(path, { method: 'PUT', body: JSON.stringify(data) }),
  patch: (path, data) => request(path, { method: 'PATCH', body: JSON.stringify(data) }),
  delete: (path) => request(path, { method: 'DELETE' }),
};
```

- [ ] **Step 7: Write `frontend/js/formFields.js`**

```js
export function escapeHtml(value) {
  return String(value ?? '').replace(/[&<>"']/g, (ch) => ({
    '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;',
  }[ch]));
}

export function renderField(field, value) {
  const val = escapeHtml(value);
  const label = escapeHtml(field.label ?? field.key);
  const required = field.required ? 'required' : '';
  const id = `field-${field.key}`;

  if (field.type === 'textarea') {
    return `<label for="${id}">${label}</label><textarea id="${id}" name="${field.key}" ${required}>${val}</textarea>`;
  }
  return `<label for="${id}">${label}</label><input id="${id}" name="${field.key}" type="text" value="${val}" ${required}>`;
}
```

  <label for="event-select">Event</label>
  <select id="event-select" name="eventId" required></select>

<script type="module">
import { api } from '/js/api.js';
import { renderField } from '/js/formFields.js';

const eventSelect = document.getElementById('event-select');
const dynamicFields = document.getElementById('dynamic-fields');
const form = document.getElementById('character-form');
const message = document.getElementById('message');
const listBody = document.querySelector('#character-list tbody');
const formTitle = document.getElementById('form-title');

let events = [];
let editingCharacterId = null;

function renderSchemaFields(schema, data = {}) {
  dynamicFields.innerHTML = schema.map((field) => renderField(field, data[field.key])).join('');
}

eventSelect.addEventListener('change', () => {
  const event = events.find((e) => e.id === eventSelect.value);
  renderSchemaFields(event ? event.character_form_schema : []);
});

async function loadEvents() {
  events = await api.get('/events');
  eventSelect.innerHTML = events.map((e) => `<option value="${e.id}">${e.name} (${e.event_date})</option>`).join('');
  if (events.length > 0) {
    renderSchemaFields(events[0].character_form_schema);
  }
}

async function loadCharacters() {
  const characters = await api.get('/characters');
  listBody.innerHTML = characters.map((c) => {
    const event = events.find((e) => e.id === c.event_id);
    return `<tr>
      <td>${c.name}</td>
      <td>${event ? event.name : c.event_id}</td>
      <td><button type="button" data-edit="${c.id}">Bearbeiten</button></td>
    </tr>`;
  }).join('');

  listBody.querySelectorAll('[data-edit]').forEach((button) => {
    button.addEventListener('click', () => startEdit(button.dataset.edit, characters));
  });
}

function startEdit(characterId, characters) {
  const character = characters.find((c) => c.id === characterId);
  if (!character) return;
  editingCharacterId = characterId;
  formTitle.textContent = `Charakter bearbeiten: ${character.name}`;
  eventSelect.value = character.event_id;
  eventSelect.disabled = true;
  form.elements.name.value = character.name;
  const event = events.find((e) => e.id === character.event_id);
  renderSchemaFields(event ? event.character_form_schema : [], character.data);
  form.querySelector('button[type="submit"]').textContent = 'Änderungen speichern';
}

function resetForm() {
  editingCharacterId = null;
  formTitle.textContent = 'Neuen Charakter anlegen';
  eventSelect.disabled = false;
  form.reset();
  form.querySelector('button[type="submit"]').textContent = 'Speichern';
  const event = events.find((e) => e.id === eventSelect.value);
  renderSchemaFields(event ? event.character_form_schema : []);
}

form.addEventListener('submit', async (event) => {
  event.preventDefault();
  message.textContent = '';
  message.className = '';
  const formData = new FormData(form);
  const { eventId, name, ...rest } = Object.fromEntries(formData);
  try {
    if (editingCharacterId) {
      await api.put(`/characters/${editingCharacterId}`, { name, data: rest });
    } else {
      await api.post('/characters', { eventId, name, data: rest });
    }
    message.textContent = 'Gespeichert.';
    message.className = 'success';
    resetForm();
    await loadCharacters();
  } catch (err) {
    message.textContent = err.status === 400 && err.body?.details
      ? err.body.details.join(', ')
      : err.message;
    message.className = 'error';
  }
});

document.getElementById('logout-link').addEventListener('click', async (evt) => {
  evt.preventDefault();
  await api.post('/auth/logout', {});
  window.location.href = '/login.html';
});

try {
  await loadEvents();
  await loadCharacters();
} catch (err) {
  if (err.status === 401) window.location.href = '/login.html';
}
</script>
</body>
</html>
